api: "GET https://sl-sd-api.smartlead.ai/api/v1/smart-delivery/runs/{id}/emails"
description: "Per-email results of a run, with authentication and bounce details"
---

<Note>
  **Smart Delivery v2 (Deliverability Test) API**: the deliverability (inbox placement) test that sends your campaign email to seed inboxes and reports where it landed. Contact support@smartlead.ai for access.
</Note>

## Overview

One item per test email, in ID order. Filter by provider, sender or placement and page with `nextCursor`.

## Path Parameters

<ParamField path="id" type="string" required>
  Run ID (UUID)
</ParamField>

## Query Parameters

<ParamField query="api_key" type="string" required>
  Your SmartLead API key
</ParamField>

<ParamField query="provider" type="string">
  Provider code, for example `GOOGLE_WORKSPACE`
</ParamField>

<ParamField query="senderId" type="number">
  Smartlead email account ID of the sender
</ParamField>

<ParamField query="placement" type="string">
  `INBOX`, `TAB`, `SPAM`, `MISSING`, `UNREADABLE`, `NOT_SENT`, `UNCONFIRMED`, `PENDING` or `BOUNCED`
</ParamField>

<ParamField query="limit" type="number">
  Page size, 1 to 200. Default 50
</ParamField>

<ParamField query="cursor" type="string">
  `nextCursor` from the previous page
</ParamField>

## Response Fields

<ResponseField name="items" type="array">
  <Expandable title="email">
    <ResponseField name="assignmentId" type="number">Email ID</ResponseField>
    <ResponseField name="provider" type="string">Provider code</ResponseField>
    <ResponseField name="senderEmailAccountId" type="number">Smartlead email account ID of the sender</ResponseField>
    <ResponseField name="senderEmail" type="string">Sending mailbox</ResponseField>
    <ResponseField name="placement" type="string">Placement</ResponseField>
    <ResponseField name="sentAt" type="string | null">Sent time</ResponseField>
    <ResponseField name="tab" type="string | null">Inbox tab it landed in, where the provider has tabs</ResponseField>
    <ResponseField name="receivedAt" type="string | null">When the copy was received</ResponseField>
    <ResponseField name="timeToInboxMs" type="number | null">Send-to-inbox time</ResponseField>
    <ResponseField name="auth" type="object">`spf`, `dkim`, `dmarc`, `arc`, `compauth` and `dmarcAligned` results of the received copy</ResponseField>
    <ResponseField name="senderIp" type="string | null">IP the email was sent from</ResponseField>
    <ResponseField name="senderPtr" type="string | null">Reverse DNS name of that IP</ResponseField>
    <ResponseField name="fcrdnsOk" type="boolean | null">Whether the reverse DNS name resolves back to the IP</ResponseField>
    <ResponseField name="sizeBytes" type="number | null">Message size</ResponseField>
    <ResponseField name="hasListUnsubscribe" type="boolean | null">Whether a List-Unsubscribe header is present</ResponseField>
    <ResponseField name="microsoft" type="object | null">Microsoft filter verdict (`scl`, `bcl`, `verdict`, `category`). `null` for copies not received at Microsoft</ResponseField>
    <ResponseField name="captured" type="boolean">Whether a copy of the received email is stored</ResponseField>
    <ResponseField name="bounce" type="object | null">`cause`, `status` and `reason` when the email bounced</ResponseField>
  </Expandable>
</ResponseField>
<ResponseField name="nextCursor" type="string | null">Pass as `cursor` for the next page. `null` on the last page</ResponseField>

## Response Codes

<ResponseField name="200" type="Success">
  Success
</ResponseField>

<ResponseField name="404" type="Error">
  Run not found (`RUN_NOT_FOUND`)
</ResponseField>

<ResponseField name="401" type="Unauthorized">
  Missing, invalid or expired API key (`UNAUTHORIZED`)
</ResponseField>

<ResponseField name="403" type="Forbidden">
  Client (agency) API keys are not allowed (`CLIENT_NOT_ALLOWED`), or your plan does not include API access (`API_ACCESS_RESTRICTED`)
</ResponseField>

<ResponseField name="429" type="Too Many Requests">
  Too many requests for this API key. Wait the number of seconds in the `Retry-After` header (`RATE_LIMITED`)
</ResponseField>
